from tiba import planning, bezier
import os
from polygonsoup import plut
from matplotlib.gridspec import GridSpec
plt = plut.plt

files = [
         ['tiba_complex_C2_03.json','spock_new_guess_05rw.json', 'spock_new_C2_01.json'],
         ]
#files = [['alfok1.json']]

nrows = len(files)
ncols = len(files[0])
print(nrows, ncols)
gs = GridSpec(nrows, ncols)
plut.figure((15,15))
for i, row in enumerate(files):
    for j, col in enumerate(row):
        plt.subplot(gs[i,j])
        planner = planning.TrajectoryPlanner(itiba)
        planner.load_paths(os.path.join(os.path.expanduser('~/Dropbox/develop_box/py5sketches/tiba_tests/outputs'),
                                        files[i][j]))
        for Xw in planner.draft:
            stroke = bezier.thick_bezier_curve(Xw[:,:2], Xw[:,2])
            plut.fill(stroke, 'k')
        plut.setup()
plt.tight_layout()
plt.show()
